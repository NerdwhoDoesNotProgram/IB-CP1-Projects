FUNCTION length (password)
    RETURN the length of password is less than 8

FUNCTION uppercase (password)
    FOR character in password
        IF 65 is greater than or equal to ordinal of characters, and is less than or equal to  90
            THEN RETURN True
        RETURN False
    END FOR

FUNCTION lowercase (password)
    FOR character in password
        IF 97 is greater than or equal to ordinal of characters, and is less than or equal to  122
            THEN RETURN True
        Return False
    END FOR

FUNCTION numbers (password)
    FOR character in password
        IF 48 is greater than or equal to ordinal of characters, and is less than or equal to  57
            THEN RETURN True
        RETURN False
    END FOR

FUNCTION special_char (password)
    SET special_chars to "!@#$%^&*()_+-=[]{}|;:,.<>?"
    FOR character in password
        IF character is in special_chars
            THEN RETURN True
        RETURN False
    END FOR
